import streamlit as st import hashlib import requests from mnemonic import Mnemonic import bip32utils st.set_page_config(page_title="VKT Blue Wallet Checker", page_icon="🔵") st.title("🔵 Blue Wallet - Seed Checker") st.warning("Apne hi words use karna, kisi aur ke words kabhi yahan mat dalna!") # Input words = st.text_area("12 Words (seed phrase) yahan dalo", placeholder="abandon abandon abandon ...") if st.button("🔍 Address Banao & Check Karo"): if not words: st.error("Pehle 12 words dalo") else: try: # 1. Words se Seed banao mnemo = Mnemonic("english") seed = mnemo.to_seed(words.strip()) # 2. Seed se BTC Address banao (BIP84 - Blue Wallet wala) root_key = bip32utils.BIP32Key.fromEntropy(seed) # m/84'/0'/0'/0/0 = Blue Wallet ka rasta child = root_key.ChildKey(84 + bip32utils.BIP32_HARDEN)\ .ChildKey(0 + bip32utils.BIP32_HARDEN)\ .ChildKey(0 + bip32utils.BIP32_HARDEN)\ .ChildKey(0).ChildKey(0) address = child.Address() st.success("Address Ban Gaya!") st.code(address) # 3. Check karo Blockchain pe account hai ya nahi st.write("Checking Blockchain...") url = f"https://blockstream.info/api/address/{address}" r = requests.get(url) if r.status_code == 200: data = r.json() tx_count = data['chain_stats']['tx_count'] + data['mempool_stats']['tx_count'] funded = data['chain_stats']['funded_txo_sum'] if tx_count == 0: st.info(f"❌ Is address par abhi tak koi BTC account / transaction nahi hai. (Fresh Address)") else: balance_sats = data['chain_stats']['funded_txo_sum'] - data['chain_stats']['spent_txo_sum'] balance_btc = balance_sats / 100000000 st.success(f"✅ Account Mojood Hai!") st.metric("Total Transactions", tx_count) st.metric("Balance", f"{balance_btc} BTC") else: st.error("API Error, dubara try karo") except Exception as e: st.error(f"Words galat hain! Error: {e}") # requirements.txt me ye dalo: # streamlit # mnemonic # bip32utils # requests
